package p

func show(arg string) {
	match lookup(arg) {
	case Ok(Some(u)):
		fmt.Println(u.Name)
	case Err(&NotFound{ID: id}):
		fmt.Println(id)
	case Err(e) if errors.Is(e, strconv.ErrSyntax):
		fmt.Println(e)
	default:
	}
}
