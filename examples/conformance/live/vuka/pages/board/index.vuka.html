package board import ( "context" "fmt" "strings" "time" "example.com/app/notes" "github.com/vuka-lang/ui" "github.com/vuka-lang/ui/validate" "github.com/vuka-lang/web" Board "../../board.ts" ) // Board is the notes, filtered by a query the URL keeps; its component is // board.ts. type Board struct { ui.Live Q string `json:"q"` @ui.Bind{Debounce: 200 * time.Millisecond} @ui.URL("q") @validate.Max(20) Rows []notes.Note `json:"rows"` @ui.Defer("notes") Total int `json:"total"` @ui.Defer("notes") Seen int `json:"seen"` Sort string `json:"sort"` @ui.URL("sort") notes *notes.Notes } @ui.Mount func (b *Board) mount(ctx context.Context) error { b.Subscribe("notes") if b.Sort == "" { b.Sort = "new" } return nil } @ui.Loads("notes") func (b *Board) loadNotes(ctx context.Context) error { b.Rows, b.Total = b.notes.Matching(b.Q), b.notes.Count() return nil } @ui.OnChange func (b *Board) changed(ctx context.Context, c ui.Changed) error { b.Rows = b.notes.Matching(b.Q) return nil } // Update takes a patch of the URL. func (b *Board) Update(p notes.BoardParams) { b.Q = p.Q b.Rows = b.notes.Matching(b.Q) } // Update takes a note a page added. func (b *Board) Update(a notes.Added) { b.Seen++ b.Rows = b.notes.Matching(b.Q) b.Total = b.notes.Count() } // Add adds a note, which every board hears of. func (b *Board) Add(ctx context.Context, text string) error { text = strings.TrimSpace(text) if text == "" { return web.Invalid{"text": "required"} } return web.Broadcast(ctx, "notes", notes.Added{Note: b.notes.Add(text)}) } // Open moves to a note's page. func (b *Board) Open(id int) { b.PushNavigate(fmt.Sprint("/notes/", id)) }
