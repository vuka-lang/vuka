package employers import ( "context" "example.com/folder-routes/pages/urls" "example.com/folder-routes/store" "github.com/vuka-lang/web" ) // New is the form adding an employer: GET /employers/new, and its Post // method POST /employers/new. type New struct { store *store.Store } func NewNew(s *store.Store) *New { return &New{store: s} } func (p *New) Title() string { return "New employer" } func (p *New) Post(ctx context.Context, form store.EmployerForm) (web.Redirect, error) { e := p.store.AddEmployer(form) return web.Redirect(urls.Employers.Employer(e.ID).Path()), nil }
